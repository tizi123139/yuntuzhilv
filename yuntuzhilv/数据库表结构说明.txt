================================================================================
              云途智旅 —— 数据库表结构说明书
================================================================================
数据库：travel_planner（MySQL 8.0+）
字符集：utf8mb4
排序规则：utf8mb4_general_ci
说明：所有表均包含 is_deleted 软删除字段、create_time / update_time 时间戳字段


================================================================================
1. 系统用户表（sys_user）
   记录系统注册游客、管理员账号信息
================================================================================

字段名              数据类型            约束 / 索引描述 / 示例
(Field)             (Type)              (Constraints / Index / Example)
─────────────────────────────────────────────────────────────────────────────
user_id             BIGINT UNSIGNED     PRIMARY KEY, AUTO_INCREMENT
                                        用户自增主键

username            VARCHAR(50)         NOT NULL, UNIQUE KEY uk_username
                                        登录用户名，不可重复

password            VARCHAR(100)        NOT NULL
                                        加密存储用户密码（BCrypt）

email               VARCHAR(100)        NULL
                                        用户绑定邮箱，如 "user@example.com"

phone               VARCHAR(20)         NULL, UNIQUE KEY uk_phone
                                        用户手机号，如 "13800138000"

real_name           VARCHAR(50)         NULL
                                        用户真实姓名 / 昵称

avatar              VARCHAR(512)        NULL
                                        用户头像图片访问地址

role                VARCHAR(20)         NOT NULL, DEFAULT 'user'
                                        角色：user 普通游客 / admin 管理员

preferences         JSON                NULL
                                        旅游偏好标签数组，如 ["美食","历史","自然"]

max_budget          INT                 DEFAULT 0
                                        用户人均预算上限（元）

status              TINYINT             NOT NULL, DEFAULT 1
                                        账号状态：1 正常，0 禁用

is_deleted          TINYINT             NOT NULL, DEFAULT 0
                                        软删除标记：0 未删除，1 已删除

create_time         DATETIME            NOT NULL, DEFAULT CURRENT_TIMESTAMP
                                        账号注册创建时间

update_time         DATETIME            NOT NULL, DEFAULT CURRENT_TIMESTAMP
                                        ON UPDATE CURRENT_TIMESTAMP
                                        账号信息更新时间

================================================================================

字段名              数据类型            约束 / 索引描述 / 示例
(Field)             (Type)              (Constraints / Index / Example)

description         TEXT                NULL

img_url             VARCHAR(512)        NULL

latitude            DECIMAL(10,6)       NULL

================================================================================
                          表间关系说明
================================================================================

sys_user (1) ──── (N) tb_itinerary          一个用户可以创建多个行程
sys_user (1) ──── (N) tb_booking_order      一个用户可以有多个预订订单
sys_user (1) ──── (N) tb_footprint          一个用户可以有多条足迹
sys_user (1) ──── (N) tb_feedback           一个用户可以提交多条反馈
sys_user (1) ──── (N) sys_log               一个用户可以有多条操作日志

tb_itinerary (1) ── (N) tb_itinerary_detail 一个行程包含多条每日明细
tb_itinerary (1) ── (N) tb_booking_order    一个行程可以关联多个预订订单

tb_attraction (1) ── (N) tb_itinerary_detail  一个景点可出现在多个行程明细中
tb_attraction (1) ── (N) tb_footprint         一个景点可被多个用户打卡

tb_hotel (1) ── (N) tb_itinerary_detail    一个酒店可出现在多个行程明细中
tb_hotel (1) ── (N) tb_booking_order       一个酒店可以有多个预订订单

================================================================================
                     建表 SQL 语句（可直接执行）
================================================================================

-- 1. 系统用户表
CREATE TABLE sys_user (
    user_id       BIGINT UNSIGNED  NOT NULL AUTO_INCREMENT PRIMARY KEY COMMENT '用户主键',
    username      VARCHAR(50)      NOT NULL COMMENT '登录用户名',
    password      VARCHAR(100)     NOT NULL COMMENT '加密密码(BCrypt)',
    email         VARCHAR(100)     NULL COMMENT '绑定邮箱',
    phone         VARCHAR(20)      NULL COMMENT '手机号',
    real_name     VARCHAR(50)      NULL COMMENT '真实姓名/昵称',
    avatar        VARCHAR(512)     NULL COMMENT '头像图片地址',
    role          VARCHAR(20)      NOT NULL DEFAULT 'user' COMMENT '角色: user/admin',
    preferences   JSON             NULL COMMENT '旅游偏好标签数组',
    max_budget    INT              NOT NULL DEFAULT 0 COMMENT '人均预算上限(元)',
    status        TINYINT          NOT NULL DEFAULT 1 COMMENT '状态: 1正常 0禁用',
    is_deleted    TINYINT          NOT NULL DEFAULT 0 COMMENT '软删除: 0未删 1已删',
    create_time   DATETIME         NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time   DATETIME         NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    UNIQUE KEY uk_username (username),
    UNIQUE KEY uk_phone (phone)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci COMMENT='系统用户表';

-- 2. 景点信息表
CREATE TABLE tb_attraction (
    attraction_id BIGINT UNSIGNED  NOT NULL AUTO_INCREMENT PRIMARY KEY COMMENT '景点主键',
    name          VARCHAR(100)     NOT NULL COMMENT '景点名称',
    city          VARCHAR(50)      NOT NULL COMMENT '所属城市',
    type          VARCHAR(50)      NOT NULL COMMENT '景点分类',
    price         DECIMAL(10,2)    NOT NULL DEFAULT 0.00 COMMENT '门票单价(元)',
    open_time     VARCHAR(50)      NOT NULL COMMENT '开放时间',
    address       VARCHAR(255)     NOT NULL COMMENT '详细地址',
    description   TEXT             NULL COMMENT '景点介绍',
    img_url       VARCHAR(512)     NULL COMMENT '封面图片地址',
    score         DECIMAL(2,1)     DEFAULT 5.0 COMMENT '综合评分(1.0~5.0)',
    visit_count   INT              NOT NULL DEFAULT 0 COMMENT '访问次数(热门排行)',
    longitude     DECIMAL(10,6)    NULL COMMENT '经度',
    latitude      DECIMAL(10,6)    NULL COMMENT '纬度',
    is_deleted    TINYINT          NOT NULL DEFAULT 0 COMMENT '软删除: 0未删 1已删',
    create_time   DATETIME         NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time   DATETIME         NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    INDEX idx_city (city),
    INDEX idx_type (type)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci COMMENT='景点信息表';

-- 3. 酒店信息表
CREATE TABLE tb_hotel (
    hotel_id      BIGINT UNSIGNED  NOT NULL AUTO_INCREMENT PRIMARY KEY COMMENT '酒店主键',
    name          VARCHAR(100)     NOT NULL COMMENT '酒店名称',
    destination   VARCHAR(50)      NOT NULL COMMENT '所属城市',
    star          VARCHAR(20)      NOT NULL COMMENT '星级: 二星/三星/四星/五星',
